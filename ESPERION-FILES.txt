*& Form FETCH_DATA
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text

FORM fetch_data .
  SELECT werks,
         matnr,
         dispo,
         fevor
  FROM marc INTO TABLE @DATA(lt_marc)
  WHERE werks = @p_werks
  AND matnr IN @s_matnr
  AND dispo IN @s_dispo
  AND fevor IN @s_fevor.

  IF lt_marc IS NOT INITIAL.

    SELECT matnr werks stlnr stlal FROM mast INTO TABLE lt_mast FOR ALL ENTRIES IN lt_marc WHERE werks = lt_marc-werks AND matnr = lt_marc-matnr
                                                                        AND stlal IN s_bom.
  ELSE.
    SELECT matnr werks stlnr stlal FROM mast INTO TABLE lt_mast WHERE werks = p_werks AND matnr IN s_matnr
                                                                        AND stlal IN s_bom.
  ENDIF.
  IF lt_mast IS NOT INITIAL.
    SELECT stlnr stlal bmeng FROM stko  INTO TABLE lt_stko  FOR ALL ENTRIES IN lt_mast WHERE stlnr = lt_mast-stlnr AND
                                                                                       stlal = lt_mast-stlal AND
                                                                                       loekz NE 'X' AND
                                                                                       stlst = '1'.
    SORT lt_mast BY stlnr stlal.
  ENDIF.

ENDFORM.
