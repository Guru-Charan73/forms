FORM user_command  USING p_ucomm LIKE sy-ucomm
                         ps_selfield TYPE slis_selfield.

  TYPES  : BEGIN OF ty_fields,
             fieldname(25) TYPE c,
           END OF ty_fields.

  DATA   : it_fields TYPE TABLE OF ty_fields,
           wa_fields TYPE ty_fields.

  CASE p_ucomm.
    WHEN 'EXPORT'.

      TRY.
*--------------------------------------------------------------------*
* Create empty XLSX document + worksheet
*--------------------------------------------------------------------*
          DATA(lo_write_access) =
          xco_cp_xlsx=>document->empty( )->write_access( ).

          DATA(lo_worksheet) =
          lo_write_access->get_workbook(
          )->worksheet->at_position( 1 ).

*--------------------------------------------------------------------*
* HEADER LABELS
*--------------------------------------------------------------------*
          DATA(lo_cursor) =
          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'A' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 1 ) ).
          lo_cursor->get_cell( )->value->write_from( 'Traceability:' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'A' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 2 )
          )->get_cell( )->value->write_from( 'CTS part number:' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'D' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 1 )
          )->get_cell( )->value->write_from( 'Materialist:' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'D' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 2 )
          )->get_cell( )->value->write_from( 'Customer PN:' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'G' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 1 )
          )->get_cell( )->value->write_from( 'Shift:' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'I' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 1 )
          )->get_cell( )->value->write_from( 'Date:' ).

*--------------------------------------------------------------------*
* HEADER VALUE PLACEHOLDERS
*--------------------------------------------------------------------*
          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'B' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 1 )
          )->get_cell( )->value->write_from( ' ' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'B' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 2 )
          )->get_cell( )->value->write_from( ' ' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'E' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 1 )
          )->get_cell( )->value->write_from( ' ' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'E' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 2 )
          )->get_cell( )->value->write_from( ' ' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'H' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 1 )
          )->get_cell( )->value->write_from( ' ' ).

          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'J' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 1 )
          )->get_cell( )->value->write_from( ' ' ).

*--------------------------------------------------------------------*
* TABLE HEADER (row 5)
*--------------------------------------------------------------------*
          DATA lt_header TYPE STANDARD TABLE OF string WITH EMPTY KEY.
          APPEND 'RHO part number'          TO lt_header.
          APPEND 'Component Description'    TO lt_header.
          APPEND 'Supplier Lot Description' TO lt_header.
          APPEND 'Supplier Lot'             TO lt_header.
          APPEND 'SAP Batch'                TO lt_header.
          APPEND 'Supplier Lot'             TO lt_header.
          APPEND 'SAP Batch'                TO lt_header.
          APPEND 'Supplier Lot'             TO lt_header.
          APPEND 'SAP Batch'                TO lt_header.

          DATA(lv_col) = 1.
          LOOP AT lt_header INTO DATA(lv_header).
            DATA(lv_column) = xco_cp_xlsx=>coordinate->for_numeric_value( lv_col ).
            lo_worksheet->cursor(
            io_column = lv_column
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( 5 )
            )->get_cell( )->value->write_from( lv_header ).
            lv_col += 1.
          ENDLOOP.

*--------------------------------------------------------------------*
* DATA (from row 6)
*--------------------------------------------------------------------*
          DATA(lv_row) = 6.
          LOOP AT gt_final ASSIGNING FIELD-SYMBOL(<fs_final>).

            lo_worksheet->cursor(
            io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'A' )
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_row )
            )->get_cell( )->value->write_from( <fs_final>-idnrk ).

            lo_worksheet->cursor(
            io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'B' )
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_row )
            )->get_cell( )->value->write_from( <fs_final>-potx1 ).

            lo_worksheet->cursor(
            io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'C' )
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_row )
            )->get_cell( )->value->write_from( ' ' ).

            lo_worksheet->cursor(
            io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'D' )
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_row )
            )->get_cell( )->value->write_from( ' ' ).

            lo_worksheet->cursor(
            io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'E' )
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_row )
            )->get_cell( )->value->write_from( ' ' ).

            lo_worksheet->cursor(
            io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'F' )
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_row )
            )->get_cell( )->value->write_from( ' ' ).

            lo_worksheet->cursor(
            io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'G' )
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_row )
            )->get_cell( )->value->write_from( ' ' ).

            lo_worksheet->cursor(
            io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'H' )
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_row )
            )->get_cell( )->value->write_from( ' ' ).

            lo_worksheet->cursor(
            io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'I' )
            io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_row )
            )->get_cell( )->value->write_from( ' ' ).

            lv_row += 1.
          ENDLOOP.

*--------------------------------------------------------------------*
* COMMENTS
*--------------------------------------------------------------------*
          DATA(lv_comment_row) = lv_row + 1.
          lo_worksheet->cursor(
          io_column = xco_cp_xlsx=>coordinate->for_alphabetic_value( 'A' )
          io_row    = xco_cp_xlsx=>coordinate->for_numeric_value( lv_comment_row )
          )->get_cell( )->value->write_from( 'Comments' ).

*--------------------------------------------------------------------*
* Get raw file content
*--------------------------------------------------------------------*
          DATA(lv_file_content) = lo_write_access->get_file_content( ).

*--------------------------------------------------------------------*
* PATCH sheet1.xml : column widths + borders + bold labels
*--------------------------------------------------------------------*
          DATA(lo_zip) = NEW cl_abap_zip( ).
          lo_zip->load( lv_file_content ).

          DATA lv_sheet_x TYPE xstring.
          lo_zip->get(
            EXPORTING name    = 'xl/worksheets/sheet1.xml'
            IMPORTING content = lv_sheet_x ).
          DATA(lv_sheet_s) = cl_abap_conv_codepage=>create_in( )->convert( lv_sheet_x ).

*-- Column widths --------------------------------------------------*
          DATA(lv_cols) =
            |<cols>|                                            &&
            |<col min="1" max="1" width="18" customWidth="1"/>| &&
            |<col min="2" max="2" width="26" customWidth="1"/>| &&
            |<col min="3" max="3" width="26" customWidth="1"/>| &&
            |<col min="4" max="4" width="14" customWidth="1"/>| &&
            |<col min="5" max="5" width="12" customWidth="1"/>| &&
            |<col min="6" max="6" width="14" customWidth="1"/>| &&
            |<col min="7" max="7" width="12" customWidth="1"/>| &&
            |<col min="8" max="8" width="14" customWidth="1"/>| &&
            |<col min="9" max="9" width="12" customWidth="1"/>| &&
            |</cols>|.

          REPLACE FIRST OCCURRENCE OF '<sheetData'
            IN lv_sheet_s WITH lv_cols && '<sheetData'.

*-- Borders on whole block (row 1 .. comment row) : s="0" -> s="1" --*
          DATA: lv_br   TYPE i,
                lv_ref  TYPE string,
                lv_find TYPE string,
                lv_add  TYPE string,
                lv_lref TYPE string.
          DATA lt_bcols TYPE TABLE OF string.
          APPEND 'A' TO lt_bcols. APPEND 'B' TO lt_bcols. APPEND 'C' TO lt_bcols.
          APPEND 'D' TO lt_bcols. APPEND 'E' TO lt_bcols. APPEND 'F' TO lt_bcols.
          APPEND 'G' TO lt_bcols. APPEND 'H' TO lt_bcols. APPEND 'I' TO lt_bcols.

          lv_br = 1.
          WHILE lv_br <= lv_comment_row.
            LOOP AT lt_bcols INTO DATA(lv_bcol).
              lv_ref  = |{ lv_bcol }{ lv_br }|.
              lv_find = '<c r="' && lv_ref && '" s="0"'.
              lv_add  = '<c r="' && lv_ref && '" s="1"'.
              REPLACE ALL OCCURRENCES OF lv_find IN lv_sheet_s WITH lv_add.
            ENDLOOP.
            lv_br = lv_br + 1.
          ENDWHILE.

*-- Bold the label cells : s="1" -> s="3" (bold + border) -----------*
          DATA lt_lbl TYPE TABLE OF string.
          APPEND 'A1'  TO lt_lbl.   " Traceability:
          APPEND 'D1'  TO lt_lbl.   " Materialist:
          APPEND 'G1'  TO lt_lbl.   " Shift:
          APPEND 'I1'  TO lt_lbl.   " Date:
          APPEND 'A2'  TO lt_lbl.   " CTS part number:
          APPEND 'D2'  TO lt_lbl.   " Customer PN:
          lv_lref = |A{ lv_comment_row }|.
          APPEND lv_lref TO lt_lbl. " Comments

          LOOP AT lt_lbl INTO DATA(lv_one).
            REPLACE ALL OCCURRENCES OF '<c r="' && lv_one && '" s="1"'
              IN lv_sheet_s WITH '<c r="' && lv_one && '" s="3"'.
          ENDLOOP.

          DATA(lv_sheet_new) = cl_abap_conv_codepage=>create_out( )->convert( lv_sheet_s ).
          lo_zip->delete( name = 'xl/worksheets/sheet1.xml' ).
          lo_zip->add(    name = 'xl/worksheets/sheet1.xml' content = lv_sheet_new ).

*--------------------------------------------------------------------*
* PATCH styles.xml : fonts(normal,bold) + fills + borders + 4 styles
*   index 1 = border | 2 = bold | 3 = bold + border
*--------------------------------------------------------------------*
          DATA lv_styles_x TYPE xstring.
          lo_zip->get(
            EXPORTING name    = 'xl/styles.xml'
            IMPORTING content = lv_styles_x ).
          DATA(lv_styles_s) = cl_abap_conv_codepage=>create_in( )->convert( lv_styles_x ).

          REPLACE FIRST OCCURRENCE OF
            '<fonts count="1"><font><sz val="11"/><name val="Calibri"/></font></fonts>'
            IN lv_styles_s WITH
            '<fonts count="2"><font><sz val="11"/><name val="Calibri"/></font>' &&
            '<font><b/><sz val="11"/><name val="Calibri"/></font></fonts>'.

          REPLACE FIRST OCCURRENCE OF
            '<fills count="1"><fill><patternFill patternType="none"/></fill></fills>'
            IN lv_styles_s WITH
            '<fills count="2"><fill><patternFill patternType="none"/></fill>' &&
            '<fill><patternFill patternType="gray125"/></fill></fills>'.

          REPLACE FIRST OCCURRENCE OF '</borders>' IN lv_styles_s WITH
            '<border><left style="thin"/><right style="thin"/>' &&
            '<top style="thin"/><bottom style="thin"/><diagonal/></border></borders>'.
          REPLACE FIRST OCCURRENCE OF '<borders count="1"' IN lv_styles_s
            WITH '<borders count="2"'.

          REPLACE FIRST OCCURRENCE OF '</cellXfs>' IN lv_styles_s WITH
            '<xf numFmtId="0" fontId="0" fillId="0" borderId="1" xfId="0" applyBorder="1"/>' &&
            '<xf numFmtId="0" fontId="1" fillId="0" borderId="0" xfId="0" applyFont="1"/>' &&
            '<xf numFmtId="0" fontId="1" fillId="0" borderId="1" xfId="0" applyFont="1" applyBorder="1"/>' &&
            '</cellXfs>'.
          REPLACE FIRST OCCURRENCE OF '<cellXfs count="1"' IN lv_styles_s
            WITH '<cellXfs count="4"'.

          DATA(lv_styles_new) = cl_abap_conv_codepage=>create_out( )->convert( lv_styles_s ).
          lo_zip->delete( name = 'xl/styles.xml' ).
          lo_zip->add(    name = 'xl/styles.xml' content = lv_styles_new ).

          lv_file_content = lo_zip->save( ).

*--------------------------------------------------------------------*
* XSTRING -> BINARY
*--------------------------------------------------------------------*
          DATA: lt_binary TYPE solix_tab,
                lv_size   TYPE i.

          CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
            EXPORTING
              buffer        = lv_file_content
            IMPORTING
              output_length = lv_size
            TABLES
              binary_tab    = lt_binary.

*--------------------------------------------------------------------*
* SAVE DIALOG + DOWNLOAD
*--------------------------------------------------------------------*
          DATA: lv_filename TYPE string,
                lv_path     TYPE string,
                lv_fullpath TYPE string.

          cl_gui_frontend_services=>file_save_dialog(
          EXPORTING
           default_extension = 'xlsx'
           default_file_name = 'Traceability.xlsx'
           file_filter       = 'Excel Files (*.xlsx)|*.xlsx|All Files (*.*)|*.*|'
          CHANGING
           filename = lv_filename
           path     = lv_path
           fullpath = lv_fullpath
          EXCEPTIONS
           OTHERS   = 1 ).

          IF sy-subrc <> 0 OR lv_fullpath IS INITIAL.
            RETURN.
          ENDIF.

          cl_gui_frontend_services=>gui_download(
          EXPORTING
            filename     = lv_fullpath
            filetype     = 'BIN'
            bin_filesize = lv_size
          CHANGING
            data_tab     = lt_binary
          EXCEPTIONS
            OTHERS       = 1 ).

          IF sy-subrc = 0.
            MESSAGE | File Downloaded: { lv_fullpath }| TYPE 'S'.
          ELSE.
            MESSAGE 'Excel download failed' TYPE 'E'.
          ENDIF.

        CATCH cx_root INTO DATA(lx_error).
          MESSAGE lx_error->get_text( ) TYPE 'E'.
      ENDTRY.

  ENDCASE.
ENDFORM.
